local function CriarTween(objeto, propriedades, tempo)
    local tweenInfo = TweenInfo.new(tempo or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local tween = TweenService:Create(objeto, tweenInfo, propriedades)
    tween:Play()
    return tween
end

local function AnimarBotao(botao, corNormal, corHover)
    botao.BackgroundColor3 = corNormal
    botao.MouseEnter:Connect(function() CriarTween(botao, {BackgroundColor3 = corHover}) end)
    botao.MouseLeave:Connect(function() CriarTween(botao, {BackgroundColor3 = corNormal}) end)
    -- Suporte extra para toque no Mobile
    botao.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            CriarTween(botao, {BackgroundColor3 = corHover}, 0.1)
        end
    end)
    botao.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            CriarTween(botao, {BackgroundColor3 = corNormal}, 0.3)
        end
    end)
end

local dragging, dragInput, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true

        end)
    end
end)
TopBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local function ConverterParaNumero(mpsString)
    if type(mpsString) == "number" then return mpsString end
    local num = tonumber(string.match(mpsString or "", "[%d%.]+")) or 0
    local sulfixo = string.match(string.upper(mpsString or ""), "%a+")
    
    if sulfixo == "K" then num = num * 1e3
    elseif sulfixo == "M" then num = num * 1e6
    elseif sulfixo == "B" then num = num * 1e9
    elseif sulfixo == "T" then num = num * 1e12
    elseif sulfixo == "Q" or sulfixo == "QD" then num = num * 1e15
    end
    return num
end

local ultimoTimestamp = 0

local function checarNuvem()
    task.spawn(function()
        pcall(function()
            local url = "https://api.jsonbin.io/v3/b/" .. BIN_ID .. "/latest"
            local response = HttpService:RequestAsync({
                Url = url,
                Method = "GET",
                Headers = {["X-Master-Key"] = API_KEY}
            })
            
            if response.Success then
                local decoded = HttpService:JSONDecode(response.Body)
                local record = decoded.record
                
                if record and record.Timestamp and record.Timestamp > ultimoTimestamp then
                    ultimoTimestamp = record.Timestamp
                    
                    if record.Pets then
                        for _, pet in ipairs(record.Pets) do
                            local currentJobId = pet.JobId or record.JobId
                            CriarNotificacao(pet.Title, pet.MPS, pet.Mutation, currentJobId)
                            task.wait(0.1) -- Suaviza a criação da lista
                        end
                    elseif record.Title then 
                        CriarNotificacao(record.Title, record.MPS, record.Mutation, record.JobId)
                    end
                end
            end
        end)
    end)
end
